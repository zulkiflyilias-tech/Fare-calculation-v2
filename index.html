// ========================================== // CONFIGURATION & API KEYS // ========================================== const HERE_API_KEY = "YOUR_HERE_API_KEY"; // Replace with your actual HERE API Key // Fare Config const BASE_FARE = 5.00; const RATE_PER_KM = 1.156; const TRAFFIC_INCENTIVE = 3.00; const OUT_OF_ZONE_SURCHARGE = 0.03; let map, routeLayer; let originCoords = null; let destCoords = null; // ========================================== // INITIALIZE MAP // ========================================== document.addEventListener("DOMContentLoaded", () => { // Center near Klang / Meru map = L.map("map").setView([3.1390, 101.4420], 12); // Tile layer (HERE Tiles or standard CartoDB/OSM tiles for base map visual) L.tileLayer("https://{s}.basemaps.cartocdn.com/rastertiles/voyager/{z}/{x}/{y}{r}.png", { maxZoom: 19, attribution: '© OpenStreetMap contributors © CARTO' }).addTo(map); }); // ========================================== // HERE GEOCODING & AUTOCOMPLETE // ========================================== async function searchLocation(query) { if (!query || query.length < 3) return []; // Center around Meru/Klang (lat: 3.13, lng: 101.44) for localized autosuggest const url = `https://autosuggest.search.hereapi.com/v1/autosuggest?q=${encodeURIComponent(query)}&at=3.1390,101.4420&apiKey=${HERE_API_KEY}&in=countryCode:MYS`; try { const res = await fetch(url); const data = await res.json(); return data.items.filter(item => item.position).map(item => ({ title: item.title, lat: item.position.lat, lng: item.position.lng })); } catch (err) { console.error("HERE Autosuggest error:", err); return []; } } // Set Origin / Destination from search or inputs function selectOrigin(lat, lng, addressName) { originCoords = { lat, lng }; document.getElementById("originInput").value = addressName; checkAndCalculateRoute(); } function selectDestination(lat, lng, addressName) { destCoords = { lat, lng }; document.getElementById("destInput").value = addressName; checkAndCalculateRoute(); } // ========================================== // HERE ROUTING & TOLL CALCULATION (v8) // ========================================== function checkAndCalculateRoute() { if (originCoords && destCoords) { calculateHereRoute(originCoords, destCoords); } } async function calculateHereRoute(origin, dest) { // Construct HERE Routing v8 API request with Tolls and Flexible Polyline requested const routingUrl = `https://router.hereapi.com/v8/routes?transportMode=car&origin=${origin.lat},${origin.lng}&destination=${dest.lat},${dest.lng}&return=summary,polyline,tolls¤cy=MYR&apiKey=${HERE_API_KEY}`; try { const response = await fetch(routingUrl); const data = await response.json(); if (!data.routes || data.routes.length === 0) { alert("No route found between these locations."); return; } const route = data.routes[0]; const section = route.sections[0]; // 1. Distance & Duration const distanceKm = section.summary.length / 1000; // meters to km const durationMins = Math.round(section.summary.duration / 60); // seconds to mins // 2. Extract Toll Cost directly from HERE API let tollCostMYR = 0; if (section.tolls && section.tolls.length > 0) { section.tolls.forEach(tollSystem => { if (tollSystem.fares) { tollSystem.fares.forEach(fare => { tollCostMYR += fare.price.value; }); } }); } // 3. Fallback i-City / Local Toll check if HERE route doesn't tag regional toll plaza const originText = (document.getElementById("originInput").value || "").toLowerCase(); const destText = (document.getElementById("destInput").value || "").toLowerCase(); if ((originText.includes("i-city") || destText.includes("i-city")) && tollCostMYR === 0) { tollCostMYR = 1.00; // Plaza Tol Suria / Bukit Raja } // 4. Draw Polyline on Map drawRouteOnMap(section.polyline); // 5. Calculate & Update UI updateFareUI(distanceKm, durationMins, tollCostMYR); } catch (error) { console.error("Error fetching HERE Route:", error); alert("Failed to calculate route using HERE API."); } } // ========================================== // FARE COMPUTATION & UI UPDATE // ========================================== function updateFareUI(distanceKm, durationMins, tollCost) { const distanceFare = distanceKm * RATE_PER_KM; const rawTotal = BASE_FARE + distanceFare + tollCost + TRAFFIC_INCENTIVE + OUT_OF_ZONE_SURCHARGE; // Round to nearest RM 0.50 const roundedTotal = Math.round(rawTotal * 2) / 2; // Render Breakdown document.getElementById("distanceText").innerText = `${distanceKm.toFixed(1)} km`; document.getElementById("durationText").innerText = `${durationMins} mins`; document.getElementById("baseFare").innerText = `RM ${BASE_FARE.toFixed(2)}`; document.getElementById("distFare").innerText = `RM ${distanceFare.toFixed(2)}`; document.getElementById("tollFare").innerText = `RM ${tollCost.toFixed(2)}`; document.getElementById("trafficFare").innerText = `RM ${TRAFFIC_INCENTIVE.toFixed(2)}`; document.getElementById("outOfZoneFare").innerText = `RM ${OUT_OF_ZONE_SURCHARGE.toFixed(2)}`; document.getElementById("finalTotal").innerText = `RM ${roundedTotal.toFixed(2)}`; document.getElementById("headerTotal").innerText = `RM ${Math.round(roundedTotal)}`; } // ========================================== // DRAW HERE POLYLINE ON LEAFLET // ========================================== function drawRouteOnMap(flexiblePolylineStr) { if (routeLayer) { map.removeLayer(routeLayer); } // Decode HERE flexible polyline string to array of [lat, lng] const decodedCoordinates = decodeHerePolyline(flexiblePolylineStr); routeLayer = L.polyline(decodedCoordinates, { color: '#8A2BE2', // Matching your purple theme weight: 6, opacity: 0.8 }).addTo(map); map.fitBounds(routeLayer.getBounds(), { padding: [40, 40] }); } // Flexible Polyline Decoder snippet for HERE Routing v8 function decodeHerePolyline(encoded) { // Basic polyline decoder for HERE v8 flexible polyline format // Returns Array of [lat, lng] let index = 0, lat = 0, lng = 0, coordinates = []; while (index < encoded.length) { let b, shift = 0, result = 0; do { b = encoded.charCodeAt(index++) - 63; result |= (b & 0x1f) << shift; shift += 5; } while (b >= 0x20); let dlat = ((result & 1) ? ~(result >> 1) : (result >> 1)); lat += dlat; shift = 0; result = 0; do { b = encoded.charCodeAt(index++) - 63; result |= (b & 0x1f) << shift; shift += 5; } while (b >= 0x20); let dlng = ((result & 1) ? ~(result >> 1) : (result >> 1)); lng += dlng; coordinates.push([lat * 1e-5, lng * 1e-5]); } return coordinates; }
